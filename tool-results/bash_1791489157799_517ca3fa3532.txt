def asyncio_run(coro):
    """Wrapper for :func:`asyncio.run` — kept here so callers can stay sync."""
    return asyncio.run(coro)


# Shared flood tracker — instantiated once per process so cooldowns
# persist across menu operations in the same session.
_flood_tracker: FloodTracker | None = None


def _get_flood_tracker() -> FloodTracker:
    global _flood_tracker
    if _flood_tracker is None:
        _flood_tracker = FloodTracker()
    return _flood_tracker


def _clear() -> None:
    os.system("cls" if os.name == "nt" else "clear")


def _banner() -> None:
    """Print the application banner."""
    print(f"{Colors.BRIGHT_YELLOW}{'═' * 60}{Colors.RESET}")
    print(f"{Colors.BRIGHT_YELLOW}  NJADDER UNIFIED v1.2.1 — Professional Edition{Colors.RESET}")
    print(f"{Colors.BRIGHT_YELLOW}  Merged from opthree.py + ramexfour.py + ts.py{Colors.RESET}")
    print(f"{Colors.BRIGHT_YELLOW}  +5 NEW Services + LIVE Ban Checker{Colors.RESET}")
    print(f"{Colors.BRIGHT_YELLOW}{'═' * 60}{Colors.RESET}")


def _section(title: str) -> None:
    print(f"{Colors.YELLOW}{'─' * 4} {title} {'─' * max(0, 50 - len(title))}{Colors.RESET}")


def _option(num: int, text: str, color: str = Colors.GREEN) -> None:
    print(f"  {color}[{num}]{Colors.RESET}  {Colors.BRIGHT}{text}{Colors.RESET}")


def _back() -> None:
    print(f"  {Colors.RED}[00]{Colors.RESET}  {Colors.RED}Back{Colors.RESET}")


def _pause() -> None:
    try:
        input(f"{Colors.INFO} Press Enter to continue...")
    except EOFError:
        pass


def _get_choice() -> str:
    try:
        return input().strip()
    except (EOFError, KeyboardInterrupt):
        return "00"


def _safe_input_int(prompt: str, default: int = 0) -> int:
    try:
        v = input(prompt).strip()
        return int(v) if v else default
    except (ValueError, EOFError, KeyboardInterrupt):
        return default


def _safe_input_float(prompt: str, default: float = 0.0) -> float:
    try:
        v = input(prompt).strip()
        return float(v) if v else default
    except (ValueError, EOFError, KeyboardInterrupt):
        return default

def _show_status() -> None:
    """Show status of files and dedup state."""
    from pathlib import Path

    _clear()
    _banner()
    _section("Status")
    print()

    data_file = Path("data.csv")
    if data_file.exists():
        import csv as csv_mod

        with open(data_file, "r", encoding="utf-8") as f:
            count = sum(1 for _ in csv_mod.DictReader(f))
        print(f"  data.csv: {Colors.GREEN}{count} members{Colors.RESET}")
    else:
        print(f"  data.csv: {Colors.RED}not found{Colors.RESET}")

    phones = load_phones()
    print(f"  phone.csv: {Colors.GREEN}{len(phones)} accounts{Colors.RESET}")

    sessions = list(Path("sessions").glob("*.session")) if Path("sessions").exists() else []
    print(f"  sessions: {Colors.GREEN}{len(sessions)} files{Colors.RESET}")

    # PeerFlood cooldown status.
    flood = _get_flood_tracker()
    flood_summary = flood.summary()
    in_cd = flood_summary["in_cooldown"]
    total = flood_summary["total"]
    if in_cd > 0:
        print(
            f"  flooded.csv: {Colors.RED}{in_cd} account(s) in 24h cooldown{Colors.RESET} "
            f"({total} total tracked)"
        )
    elif total > 0:
        print(
            f"  flooded.csv: {Colors.GREEN}0 in cooldown{Colors.RESET} "
            f"({total} previously flooded)"
        )
    else:
        print(f"  flooded.csv: {Colors.GREEN}no PeerFlood events recorded{Colors.RESET}")

    # Dedup state.
    dedup = DedupManager()
    print(f"  done.csv: {Colors.GREEN}{dedup.count()} done{Colors.RESET}")

    print()
    _pause()
